- Маршруты с суммой дистанций

🏗️ **Условие:** Напиши функцию `routeTotals(routes, minSegments)`, которая оставляет маршруты с количеством сегментов `>= minSegments` и возвращает `{ id, total }`, где `total` — сумма `distance`. 

      // Вход:
const routes = [
  { id: 'r1', segments: [{ distance: 10 }, { distance: 15 }] },
  { id: 'r2', segments: [{ distance: 5 }] },
  { id: 'r3', segments: [{ distance: 7 }, { distance: 8 }, { distance: 10 }] }
];
const minSegments = 2;

// Вызов:
const result = routeTotals(routes, minSegments);

// Ожидаемый результат:
const expected = [
  { id: 'r1', total: 25 },
  { id: 'r3', total: 25 }
];
